import { createFileRoute, Link, Outlet, useNavigate, useRouterState } from "@tanstack/react-router";
import { useAuth } from "@/hooks/use-auth";
import { useQuery } from "@tanstack/react-query";
import { api } from "@/lib/api";
import {
  AlertTriangle,
  Archive,
  BarChart3,
  Bell,
  Calendar,
  CheckSquare,
  ChevronRight,
  DatabaseBackup,
  ExternalLink,
  FileText,
  GraduationCap,
  LayoutDashboard,
  LogOut,
  MapPin,
  Plus,
  ScrollText,
  Settings,
  ShieldCheck,
  Tags,
  UserCircle,
  Users,
  MonitorSmartphone,
  type LucideIcon,
} from "lucide-react";
import { useEffect } from "react";

type NavLinkItem = {
  to: string;
  label: string;
  icon: LucideIcon;
  end?: boolean;
  iconBg?: string;
  badge?: number;
};
type NavSection = {
  title: string | null;
  items: NavLinkItem[];
  adminOnly?: boolean;
};

export const Route = createFileRoute("/_authenticated/admin")({
  component: AdminLayout,
});

function AdminLayout() {
  const { canManageContent, isAdmin, signOut, user } = useAuth();
  const nav = useNavigate();
  const routerState = useRouterState();

  const { data: dashboard } = useQuery({
    queryKey: ["admin", "summary"],
    queryFn: () => api.admin.dashboard(),
    staleTime: 30_000,
    enabled: canManageContent,
  });

  useEffect(() => {
    if (!canManageContent) nav({ to: "/", replace: true });
  }, [canManageContent, nav]);

  if (!canManageContent) return null;

  const pendingCount = dashboard?.pending_approval_resources ?? 0;

  const NAV_SECTIONS: NavSection[] = [
    {
      title: null,
      items: [
        { to: "/admin", label: "Dashboard", icon: LayoutDashboard, end: true, iconBg: "bg-primary/80" },
      ],
    },
    {
      title: "Content",
      items: [
        { to: "/admin/content",      label: "Resources",    icon: FileText,      iconBg: "bg-blue-600/80" },
        { to: "/admin/approvals",    label: "Approvals",    icon: CheckSquare,   iconBg: "bg-amber-500/80", badge: pendingCount > 0 ? pendingCount : undefined },
        { to: "/admin/digitization", label: "Digitization", icon: Archive,       iconBg: "bg-violet-600/80" },
      ],
    },
    {
      title: "Community",
      items: [
        { to: "/admin/events",   label: "Events",      icon: Calendar,      iconBg: "bg-sky-600/80" },
        { to: "/admin/training", label: "Training",    icon: GraduationCap, iconBg: "bg-teal-600/80" },
        { to: "/admin/sites",    label: "Demo Sites",  icon: MapPin,        iconBg: "bg-rose-600/80" },
      ],
    },
    {
      title: "Analytics",
      items: [
        { to: "/admin/reports", label: "Reports", icon: BarChart3, iconBg: "bg-orange-500/80" },
      ],
    },
    {
      title: "System",
      adminOnly: true,
      items: [
        { to: "/admin/taxonomy",  label: "Categories & Types", icon: Tags,              iconBg: "bg-indigo-600/80" },
        { to: "/admin/users",     label: "Users & Roles",      icon: Users,             iconBg: "bg-cyan-600/80" },
        { to: "/admin/sessions",  label: "Sessions",           icon: MonitorSmartphone, iconBg: "bg-violet-600/80" },
        { to: "/admin/audit",     label: "Audit Logs",         icon: ScrollText,        iconBg: "bg-slate-500/80" },
        { to: "/admin/backup",    label: "Backup",             icon: DatabaseBackup,    iconBg: "bg-emerald-600/80" },
        { to: "/admin/settings",  label: "Settings",           icon: Settings,          iconBg: "bg-gray-500/80" },
      ],
    },
  ];

  const initials = user?.full_name
    ? user.full_name.split(" ").map((w) => w[0]).join("").slice(0, 2).toUpperCase()
    : (user?.email?.slice(0, 2).toUpperCase() ?? "?");

  const displayName = user?.full_name || "Administrator";
  const roleLabel = isAdmin ? "System Administrator" : "Content Manager";
  const isStaleSession = !user?.full_name;

  const currentPath = routerState.location.pathname;

  return (
    <div className="flex min-h-screen" style={{ background: "#f1f5f1" }}>

      {/* ════════════════════════════════════════
          SIDEBAR
      ════════════════════════════════════════ */}
      <aside
        className="sticky top-0 hidden h-screen w-60 shrink-0 flex-col overflow-hidden md:flex"
        style={{ background: "linear-gradient(180deg, #0c2b1a 0%, #091e12 100%)" }}
      >
        {/* Subtle radial highlight at top-left */}
        <div
          aria-hidden
          className="pointer-events-none absolute inset-0"
          style={{
            background:
              "radial-gradient(ellipse 260px 200px at 0% 0%, rgba(255,255,255,0.06) 0%, transparent 70%)",
          }}
        />

        {/* ── Brand ── */}
        <div className="relative flex shrink-0 items-center gap-3 border-b border-white/15 px-4 py-4">
          <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-white shadow-md shadow-black/30">
            <img src="/logo.png" alt="UNZA" className="h-7 w-7 object-contain" />
          </div>
          <div className="min-w-0 flex-1">
            <div className="truncate text-[13px] font-bold leading-tight text-white">Sanitation Hub</div>
            <div className="text-[10px] font-medium text-white/60 uppercase tracking-widest">UNZA · Admin</div>
          </div>
          {isAdmin && (
            <div className="flex h-5 items-center rounded-full bg-amber-400/25 px-1.5 text-[9px] font-bold uppercase tracking-wider text-amber-200">
              Admin
            </div>
          )}
        </div>

        {/* ── Quick create ── */}
        <div className="relative shrink-0 px-3 py-3">
          <Link
            to="/admin/content"
            className="flex w-full items-center justify-center gap-2 rounded-lg border border-white/20 bg-white/10 py-2 text-xs font-semibold text-white/90 transition hover:bg-white/18 hover:text-white"
          >
            <Plus className="h-3.5 w-3.5" />
            New Resource
          </Link>
        </div>

        {/* ── Stale session warning ── */}
        {isStaleSession && (
          <div className="relative mx-3 mb-2 shrink-0 flex items-start gap-2 rounded-lg border border-orange-400/25 bg-orange-500/12 px-3 py-2.5 text-[11px] text-orange-200/80">
            <AlertTriangle className="mt-0.5 h-3.5 w-3.5 shrink-0 text-orange-300" />
            <span>Session expired — please sign out and back in.</span>
          </div>
        )}

        {/* ── Navigation ── */}
        <nav className="relative flex-1 overflow-y-auto px-3 pb-3 scrollbar-thin">
          {NAV_SECTIONS.map((section) => {
            if (section.adminOnly && !isAdmin) return null;
            return (
              <div key={section.title ?? "__main"} className="mb-1">
                {section.title && (
                  <div className="mb-1 mt-4 flex items-center gap-2 px-2">
                    <span className="text-[9px] font-bold uppercase tracking-[0.2em] text-white/50">
                      {section.title}
                    </span>
                    <div className="h-px flex-1 bg-white/15" />
                  </div>
                )}
                <div className="space-y-0.5">
                  {section.items.map((item) => {
                    const isActive = item.end
                      ? currentPath === item.to
                      : currentPath.startsWith(item.to);
                    return (
                      <Link
                        key={item.to}
                        to={item.to}
                        activeOptions={{ exact: !!item.end }}
                        className={[
                          "group relative flex items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] font-medium transition-all",
                          isActive
                            ? "bg-white/15 text-white shadow-sm"
                            : "text-white/75 hover:bg-white/10 hover:text-white",
                        ].join(" ")}
                      >
                        {/* Active left bar */}
                        {isActive && (
                          <span className="absolute left-0 top-1/2 h-5 w-0.5 -translate-y-1/2 rounded-full bg-green-400" />
                        )}
                        {/* Icon */}
                        <div
                          className={[
                            "flex h-6 w-6 shrink-0 items-center justify-center rounded-md transition",
                            isActive
                              ? `${item.iconBg ?? "bg-primary/80"} shadow-sm`
                              : "bg-white/12 group-hover:bg-white/18",
                          ].join(" ")}
                        >
                          <item.icon className="h-3.5 w-3.5 text-white" />
                        </div>
                        <span className="flex-1 truncate">{item.label}</span>
                        {/* Badge */}
                        {item.badge !== undefined && item.badge > 0 && (
                          <span className="flex h-4.5 min-w-[18px] items-center justify-center rounded-full bg-amber-400 px-1 text-[10px] font-bold text-amber-900">
                            {item.badge > 99 ? "99+" : item.badge}
                          </span>
                        )}
                      </Link>
                    );
                  })}
                </div>
              </div>
            );
          })}

          {/* Profile link */}
          <div className="mt-4 border-t border-white/15 pt-3">
            <Link
              to="/profile"
              className="group flex items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] font-medium text-white/75 transition hover:bg-white/10 hover:text-white"
            >
              <div className="flex h-6 w-6 shrink-0 items-center justify-center rounded-md bg-white/12 group-hover:bg-white/18">
                <UserCircle className="h-3.5 w-3.5 text-white" />
              </div>
              My Profile
            </Link>
          </div>
        </nav>

        {/* ── User footer ── */}
        <div className="relative shrink-0 border-t border-white/15 p-3">
          <div className="flex items-center gap-2.5">
            {/* Avatar circle */}
            <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-primary text-xs font-bold text-white shadow-md ring-2 ring-white/20">
              {initials}
            </div>
            <div className="min-w-0 flex-1">
              <div className="flex items-center gap-1.5">
                <p className="truncate text-[13px] font-semibold text-white">{displayName}</p>
                {isAdmin && <ShieldCheck className="h-3 w-3 shrink-0 text-green-400" />}
              </div>
              <p className="truncate text-[10px] text-white/60">{roleLabel}</p>
            </div>
          </div>
          {/* Action row */}
          <div className="mt-2.5 grid grid-cols-2 gap-1.5">
            <Link
              to="/"
              className="flex items-center justify-center gap-1.5 rounded-lg border border-white/15 bg-white/8 py-1.5 text-[11px] font-medium text-white/75 transition hover:bg-white/15 hover:text-white"
            >
              <ExternalLink className="h-3 w-3" /> Public site
            </Link>
            <button
              onClick={() => signOut()}
              className="flex items-center justify-center gap-1.5 rounded-lg border border-white/15 bg-white/8 py-1.5 text-[11px] font-medium text-white/75 transition hover:border-rose-400/40 hover:bg-rose-500/20 hover:text-rose-200"
            >
              <LogOut className="h-3 w-3" /> Sign out
            </button>
          </div>
        </div>
      </aside>

      {/* ════════════════════════════════════════
          MAIN AREA
      ════════════════════════════════════════ */}
      <div className="flex min-w-0 flex-1 flex-col">

        {/* ── Top bar ── */}
        <header className="sticky top-0 z-30 flex h-13 shrink-0 items-center gap-3 border-b border-border bg-white/95 px-5 shadow-sm backdrop-blur">
          {/* Breadcrumb spacer — pages can fill this via portal if needed */}
          <div className="flex-1" />

          {/* Approval alert pill */}
          {pendingCount > 0 && (
            <Link
              to="/admin/approvals"
              className="hidden items-center gap-1.5 rounded-full border border-amber-200 bg-amber-50 px-3 py-1 text-xs font-semibold text-amber-700 hover:bg-amber-100 transition-colors sm:flex"
            >
              <CheckSquare className="h-3.5 w-3.5" />
              {pendingCount} pending approval{pendingCount !== 1 ? "s" : ""}
            </Link>
          )}

          {/* View public site */}
          <Link
            to="/"
            className="hidden items-center gap-1.5 rounded-md border border-border px-3 py-1.5 text-xs font-medium text-muted-foreground hover:bg-secondary transition-colors sm:flex"
          >
            <ExternalLink className="h-3.5 w-3.5" /> Public site
          </Link>

          {/* User avatar */}
          <div className="flex items-center gap-2">
            <Link
              to="/profile"
              className="flex h-8 w-8 items-center justify-center rounded-full bg-primary text-xs font-bold text-white shadow-sm ring-2 ring-primary/20 hover:ring-primary/40 transition-all"
              title={displayName}
            >
              {initials}
            </Link>
          </div>
        </header>

        {/* ── Page content ── */}
        <main className="flex-1 overflow-x-hidden">
          <Outlet />
        </main>
      </div>
    </div>
  );
}
